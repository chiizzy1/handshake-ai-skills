# Dependencies for gaffer_inspect.py
#
# Only ffmpeg is truly required. Everything else improves a stage, and the
# script reports any stage it had to skip rather than failing silently.
#
# ---------------------------------------------------------------------------
# Recommended setup: a dedicated environment on Python 3.10 or newer
# ---------------------------------------------------------------------------
#
# yt-dlp dropped Python 3.9 after 2025.10.14, and YouTube extraction breaks
# often enough that being stuck on an old yt-dlp eventually costs you downloads.
# macOS ships 3.9, so install a newer interpreter alongside it. uv does this
# without sudo:
#
#   python3 -m pip install --user uv
#   export PATH="$HOME/Library/Python/3.9/bin:$PATH"
#   uv python install 3.13
#   uv venv ~/.venvs/gaffer-tools --python 3.13
#   uv pip install --python ~/.venvs/gaffer-tools/bin/python -r requirements.txt
#
# Then run the inspector with that interpreter:
#
#   ~/.venvs/gaffer-tools/bin/python gaffer_inspect.py --check-deps
#
# Plain pip works too if you already have Python 3.10+:
#
#   python3 -m venv ~/.venvs/gaffer-tools
#   ~/.venvs/gaffer-tools/bin/pip install -r requirements.txt
#
# The script also runs on a bare system Python 3.9 with whatever is installed
# there. It just cannot use the newest yt-dlp.
#
# imageio-ffmpeg ships a bundled ffmpeg binary, so a system install is optional.

imageio-ffmpeg>=0.4.9     # ffmpeg binary: frames, audio, cut detection, loudness
opencv-python>=4.8        # faster and more accurate frame seeking
yt-dlp>=2024.1.1          # only needed when the source is a URL
openai-whisper>=20231117  # draft transcript with timestamps
numpy>=1.24               # array handling for frames and audio
